cmake_minimum_required(VERSION 3.26)
project(sogen_raster_web LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Widgets)
qt_standard_project_setup()
if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "This presenter requires the WebAssembly SDK")
endif()
qt_add_executable(raster_display host.cpp)
target_link_libraries(raster_display PRIVATE Qt6::Widgets)
target_include_directories(raster_display PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../src/qt-raster-protocol")
target_link_options(raster_display PRIVATE "-sEXPORTED_FUNCTIONS=['_main','_raster_receive','_raster_clear','_raster_release_input','_raster_release_geometry','_malloc','_free']")
set_target_properties(raster_display PROPERTIES QT_WASM_EXTRA_EXPORTED_METHODS "HEAPU8")
